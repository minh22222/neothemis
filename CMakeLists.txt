cmake_minimum_required(VERSION 3.16)

project(NeoThemis LANGUAGES CXX)

include(CTest)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NEOTHEMIS_BUILD_GUI "Build the Qt desktop GUI" ON)
option(NEOTHEMIS_BUILD_SERVER "Build the local judging web server" ON)

add_subdirectory(core)
if (NEOTHEMIS_BUILD_GUI OR NEOTHEMIS_BUILD_SERVER)
    add_subdirectory(server/database)
endif()
add_subdirectory(cli)
add_subdirectory(gui)
add_subdirectory(server)

if (BUILD_TESTING)
    add_subdirectory(tests)
endif()

if (MSVC)
    target_compile_options(neothemis_core PRIVATE /W4)
    target_compile_options(neothemis-cli PRIVATE /W4)
    if (TARGET neothemis_server_database)
        target_compile_options(neothemis_server_database PRIVATE /W4)
    endif()
    if (TARGET neothemis_server_http)
        target_compile_options(neothemis_server_http PRIVATE /W4)
    endif()
    if (TARGET neothemis-gui)
        target_compile_options(neothemis-gui PRIVATE /W4)
    endif()
    if (TARGET neothemis-server)
        target_compile_options(neothemis-server PRIVATE /W4)
    endif()
else()
    target_compile_options(neothemis_core PRIVATE -Wall -Wextra -pedantic)
    target_compile_options(neothemis-cli PRIVATE -Wall -Wextra -pedantic)
    if (TARGET neothemis_server_database)
        target_compile_options(neothemis_server_database PRIVATE -Wall -Wextra -pedantic)
    endif()
    if (TARGET neothemis_server_http)
        target_compile_options(neothemis_server_http PRIVATE -Wall -Wextra -pedantic)
    endif()
    if (TARGET neothemis-gui)
        target_compile_options(neothemis-gui PRIVATE -Wall -Wextra -pedantic)
    endif()
    if (TARGET neothemis-server)
        target_compile_options(neothemis-server PRIVATE -Wall -Wextra -pedantic)
    endif()
endif()
